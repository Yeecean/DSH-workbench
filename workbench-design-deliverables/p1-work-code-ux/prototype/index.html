src/auth/session-guard.ts · Replay Protection 
export function verifyReplayWindow(token: SessionToken, now = Date.now()): boolean {
  if (!token.nonce || !token.issuedAt) return false;
  if (seenNonces.has(token.nonce)) return false;
  seenNonces.add(token.nonce);
  return Math.abs(now - token.issuedAt) <= Math.min(token.ttlMs ?? 120_000, 180_000);
}
